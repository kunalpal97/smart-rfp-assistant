import { useQuery } from "@tanstack/react-query";
import {
  AlertCircle,
  FileText,
  Files,
  LoaderCircle,
  RefreshCw,
} from "lucide-react";

import { useAppAuth } from "../../auth/AuthContext";
import { ApiError } from "../../lib/api";

import {
  getDocuments,
  type DocumentItem,
} from "./api/documentsApi";

import DocumentUpload from "./components/DocumentUpload";
import DeleteDocumentButton from "./components/DeleteDocumentButton";

// Convert backend errors into user-friendly messages.
function getDocumentErrorMessage(error: unknown): string {
  if (error instanceof ApiError) {
    if (error.status === 401) {
      return "Your session is no longer authorized. Please sign out and sign in again.";
    }

    if (error.status === 403) {
      return "You do not have permission to view documents.";
    }

    return error.message;
  }

  if (error instanceof TypeError) {
    return "Unable to connect to the backend. Check that FastAPI is running.";
  }

  return error instanceof Error
    ? error.message
    : "Something went wrong while loading documents.";
}

// Render one document row.
function DocumentRow({
  document,
  isAdmin,
}: {
  document: DocumentItem;
  isAdmin: boolean;
}) {
  return (
    <tr className="border-t border-slate-100 hover:bg-slate-50">
      {/* Document details */}
      <td className="px-5 py-4">
        <div className="flex items-center gap-3">
          <div className="rounded-lg bg-blue-50 p-2">
            <FileText className="h-5 w-5 text-blue-600" />
          </div>

          <div className="min-w-0">
            <p className="break-all text-sm font-medium text-slate-900">
              {document.filename}
            </p>

            <p className="mt-1 text-xs text-slate-500">
              Indexed document
            </p>
          </div>
        </div>
      </td>

      {/* Chunk count */}
      <td className="px-5 py-4 text-right text-sm text-slate-600">
        {document.chunk_count}
      </td>

      {/* Admin-only actions */}
      {isAdmin && (
        <td className="px-5 py-4 text-right">
          <DeleteDocumentButton document={document} />
        </td>
      )}
    </tr>
  );
}

export default function DocumentsPage() {
  // Role comes from backend-verified AuthContext.
  const { isAdmin } = useAppAuth();

  // Fetch documents using the authenticated API client.
  const documentsQuery = useQuery({
    queryKey: ["documents"],
    queryFn: getDocuments,

    retry: (failureCount: number, error: unknown) => {
      // Do not retry authentication or permission errors.
      if (
        error instanceof ApiError &&
        [401, 403, 404].includes(error.status)
      ) {
        return false;
      }

      return failureCount < 2;
    },

    staleTime: 30_000,
  });

  const documents = documentsQuery.data?.documents ?? [];

  const totalDocuments =
    documentsQuery.data?.total_documents ?? 0;
  
  
  return (
    <section className="space-y-6">
      {/* Page Header */}
      <div className="flex flex-wrap items-center justify-between gap-4">
        <div>
          <div className="flex items-center gap-3">
            <Files className="h-8 w-8 text-blue-600" />

            <h1 className="text-3xl font-semibold text-slate-900">
              Document Library
            </h1>
          </div>

          <p className="mt-2 text-sm text-slate-500">
            Manage the historical proposals used by the AI assistant.
          </p>
        </div>

        <button
          type="button"
          onClick={() => void documentsQuery.refetch()}
          disabled={documentsQuery.isFetching}
          className="inline-flex items-center gap-2 rounded-lg border border-slate-200 bg-white px-4 py-2 text-sm font-medium text-slate-700 hover:bg-slate-50 disabled:cursor-not-allowed disabled:opacity-50"
        >
          <RefreshCw
            className={`h-4 w-4 ${
              documentsQuery.isFetching ? "animate-spin" : ""
            }`}
          />
          Refresh
        </button>
      </div>

      {/* Upload is visible only to administrators */}
      {isAdmin && <DocumentUpload />}

      {/* Document Summary */}
      <div className="rounded-xl border border-slate-200 bg-white p-5">
        <p className="text-sm text-slate-500">
          Total indexed documents
        </p>

        <p className="mt-2 text-3xl font-semibold text-slate-900">
          {documentsQuery.isPending
            ? "—"
            : documentsQuery.isError
              ? "—"
              : totalDocuments}
        </p>
      </div>

      {/* Document Table */}
      <div className="overflow-hidden rounded-xl border border-slate-200 bg-white">
        <div className="border-b border-slate-200 px-5 py-4">
          <h2 className="text-base font-semibold text-slate-900">
            Documents
          </h2>

          <p className="mt-1 text-xs text-slate-500">
            Files currently indexed in the knowledge base.
          </p>
        </div>

        {/* Loading State */}
        {documentsQuery.isPending ? (
          <div
            role="status"
            className="flex items-center justify-center gap-3 px-5 py-16 text-sm text-slate-500"
          >
            <LoaderCircle className="h-5 w-5 animate-spin" />
            Loading documents...
          </div>
        ) : documentsQuery.isError ? (
          /* Error State */
          <div role="alert" className="px-5 py-12 text-center">
            <AlertCircle className="mx-auto h-8 w-8 text-red-500" />

            <p className="mt-3 text-sm font-semibold text-slate-900">
              Unable to load documents
            </p>

            <p className="mx-auto mt-2 max-w-md text-sm text-slate-500">
              {getDocumentErrorMessage(documentsQuery.error)}
            </p>

            <button
              type="button"
              onClick={() => void documentsQuery.refetch()}
              className="mt-5 rounded-lg bg-slate-900 px-4 py-2 text-sm font-medium text-white hover:bg-slate-800"
            >
              Try again
            </button>
          </div>
        ) : documents.length === 0 ? (
          /* Empty State */
          <div className="px-5 py-16 text-center">
            <Files className="mx-auto h-10 w-10 text-slate-300" />

            <p className="mt-4 text-sm font-semibold text-slate-900">
              No documents available
            </p>

            <p className="mt-2 text-sm text-slate-500">
              Documents uploaded to the knowledge base will appear here.
            </p>
          </div>
        ) : (
          /* Documents Table */
          <div className="overflow-x-auto">
            <table className="w-full">
              <thead className="bg-slate-50">
                <tr>
                  <th
                    scope="col"
                    className="px-5 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500"
                  >
                    Document
                  </th>

                  <th
                    scope="col"
                    className="px-5 py-3 text-right text-xs font-semibold uppercase tracking-wide text-slate-500"
                  >
                    Chunks
                  </th>

                  {/* Only Admin sees the Actions column */}
                  {isAdmin && (
                    <th
                      scope="col"
                      className="px-5 py-3 text-right text-xs font-semibold uppercase tracking-wide text-slate-500"
                    >
                      Actions
                    </th>
                  )}
                </tr>
              </thead>

              <tbody>
                {documents.map((document) => (
                  <DocumentRow
                    key={document.document_id}
                    document={document}

                  />
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
    </section>
  );
}

import { useRef, useState } from "react";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { AlertCircle, CheckCircle2, Upload } from "lucide-react";

import { ApiError } from "../../../lib/api";
import { uploadDocument } from "../api/documentsApi";

const MAX_FILE_SIZE = 10 * 1024 * 1024;
const ALLOWED_EXTENSIONS = [".pdf", ".txt", ".md"];

function validateFile(file: File): string | null {
  const extension = file.name.slice(file.name.lastIndexOf(".")).toLowerCase();

  if (!ALLOWED_EXTENSIONS.includes(extension)) {
    return "Only PDF, TXT and MD files are supported.";
  }

  if (file.size === 0) {
    return "The selected file is empty.";
  }

  if (file.size > MAX_FILE_SIZE) {
    return "File size must not exceed 10 MB.";
  }

  return null;
}

function getUploadErrorMessage(error: unknown): string {
  if (error instanceof ApiError) {
    switch (error.status) {
      case 400:
        return error.message;
      case 401:
        return "Your session is no longer authorized. Please sign out and sign in again.";
      case 403:
        return "Only administrators can upload documents.";
      case 409:
        return error.message;
      case 413:
        return "File size must not exceed 10 MB.";
      case 422:
        return error.message;
      default:
        return error.message;
    }
  }

  if (error instanceof TypeError) {
    return "Unable to connect to the backend. Check that FastAPI is running.";
  }

  return error instanceof Error
    ? error.message
    : "Document upload failed. Please try again.";
}

export default function DocumentUpload() {
  const inputRef = useRef<HTMLInputElement>(null);
  const queryClient = useQueryClient();

  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [validationError, setValidationError] = useState<string | null>(null);
  const [successMessage, setSuccessMessage] = useState<string | null>(null);

  const uploadMutation = useMutation({
    mutationFn: uploadDocument,

    onSuccess: async (result) => {
      setSuccessMessage(
        `${result.original_filename} uploaded successfully. ${result.chunks_created} chunks indexed.`,
      );

      setSelectedFile(null);
      setValidationError(null);

      if (inputRef.current) {
        inputRef.current.value = "";
      }

      await queryClient.invalidateQueries({
        queryKey: ["documents"],
      });
    },
  });

  function handleFileChange(event: React.ChangeEvent<HTMLInputElement>) {
    setValidationError(null);
    setSuccessMessage(null);
    uploadMutation.reset();

    const file = event.target.files?.[0] ?? null;

    if (!file) {
      setSelectedFile(null);
      return;
    }

    const error = validateFile(file);

    if (error) {
      setSelectedFile(null);
      setValidationError(error);
      event.target.value = "";
      return;
    }

    setSelectedFile(file);
  }

  function handleUpload() {
    if (!selectedFile || uploadMutation.isPending) {
      return;
    }

    setSuccessMessage(null);
    setValidationError(null);

    uploadMutation.mutate(selectedFile);
  }

  return (
    <div className="rounded-xl border border-slate-200 bg-white p-5">
      <div className="mb-4">
        <h2 className="text-base font-semibold text-slate-900">
          Upload document
        </h2>

        <p className="mt-1 text-sm text-slate-500">
          Add a historical proposal to the AI knowledge base.
        </p>
      </div>

      <div className="flex flex-wrap items-center gap-3">
        <input
          ref={inputRef}
          type="file"
          accept=".pdf,.txt,.md"
          onChange={handleFileChange}
          disabled={uploadMutation.isPending}
          aria-label="Select document"
          className="block min-w-0 flex-1 text-sm text-slate-600 file:mr-4 file:rounded-lg file:border-0 file:bg-slate-100 file:px-4 file:py-2 file:font-medium file:text-slate-700 hover:file:bg-slate-200"
        />

        <button
          type="button"
          onClick={handleUpload}
          disabled={!selectedFile || uploadMutation.isPending}
          className="inline-flex items-center gap-2 rounded-lg bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700 disabled:cursor-not-allowed disabled:opacity-50"
        >
          <Upload className="h-4 w-4" />
          {uploadMutation.isPending ? "Uploading..." : "Upload"}
        </button>
      </div>

      <p className="mt-3 text-xs text-slate-500">
        Supported formats: PDF, TXT, MD. Maximum size: 10 MB.
      </p>

      {validationError && (
        <div
          role="alert"
          className="mt-4 flex items-start gap-2 rounded-lg bg-red-50 p-3 text-sm text-red-700"
        >
          <AlertCircle className="mt-0.5 h-4 w-4 shrink-0" />
          {validationError}
        </div>
      )}

      {uploadMutation.isError && (
        <div
          role="alert"
          className="mt-4 flex items-start gap-2 rounded-lg bg-red-50 p-3 text-sm text-red-700"
        >
          <AlertCircle className="mt-0.5 h-4 w-4 shrink-0" />
          {getUploadErrorMessage(uploadMutation.error)}
        </div>
      )}

      {successMessage && (
        <div
          role="status"
          className="mt-4 flex items-start gap-2 rounded-lg bg-green-50 p-3 text-sm text-green-700"
        >
          <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" />
          {successMessage}
        </div>
      )}
    </div>
  );
}

========== frontend-react/src/features/documents/components/DeleteDocumentButton.tsx ==========

import { useEffect, useRef, useState } from "react";
import { createPortal } from "react-dom";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { AlertCircle, LoaderCircle, Trash2 } from "lucide-react";

import { ApiError } from "../../../lib/api";
import {
  deleteDocument,
  type DocumentItem,
} from "../api/documentsApi";

type DeleteDocumentButtonProps = {
  document: DocumentItem;
};

function getDeleteErrorMessage(error: unknown): string {
  if (error instanceof ApiError) {
    switch (error.status) {
      case 401:
        return "Your session has expired. Please sign out and sign in again.";
      case 403:
        return "You do not have permission to delete documents.";
      case 404:
        return "This document no longer exists. Refresh the list.";
      default:
        return error.message;
    }
  }

  if (error instanceof TypeError) {
    return "Unable to connect to the backend. Check that FastAPI is running.";
  }

  return error instanceof Error
    ? error.message
    : "Unable to delete the document.";
}

export default function DeleteDocumentButton({
  document,
}: DeleteDocumentButtonProps) {
  const queryClient = useQueryClient();
  const [confirmOpen, setConfirmOpen] = useState(false);

  const triggerRef = useRef<HTMLButtonElement>(null);
  const cancelRef = useRef<HTMLButtonElement>(null);
  const dialogRef = useRef<HTMLDivElement>(null);

  const deleteMutation = useMutation({
    mutationFn: () => deleteDocument(document.document_id),

    onSuccess: async () => {
      setConfirmOpen(false);

      await queryClient.invalidateQueries({
        queryKey: ["documents"],
      });
    },
  });

  function openConfirmation() {
    deleteMutation.reset();
    setConfirmOpen(true);
  }

  function closeConfirmation() {
    if (!deleteMutation.isPending) {
      setConfirmOpen(false);
      triggerRef.current?.focus();
    }
  }

  useEffect(() => {
    if (!confirmOpen) return;

    const previousOverflow = documentBodyOverflow();
    window.document.body.style.overflow = "hidden";

    cancelRef.current?.focus();

    function handleKeyDown(event: KeyboardEvent) {
      if (event.key === "Escape") {
        if (!deleteMutation.isPending) {
          event.preventDefault();
          setConfirmOpen(false);
          triggerRef.current?.focus();
        }
        return;
      }

      if (event.key !== "Tab") return;

      const dialog = dialogRef.current;
      if (!dialog) return;

      const focusable = Array.from(
        dialog.querySelectorAll<HTMLButtonElement>(
          "button:not(:disabled)"
        )
      );

      if (focusable.length === 0) {
        event.preventDefault();
        return;
      }

      const first = focusable[0];
      const last = focusable[focusable.length - 1];

      if (event.shiftKey && window.document.activeElement === first) {
        event.preventDefault();
        last.focus();
      } else if (
        !event.shiftKey &&
        window.document.activeElement === last
      ) {
        event.preventDefault();
        first.focus();
      }
    }

    window.document.addEventListener("keydown", handleKeyDown);

    return () => {
      window.document.body.style.overflow = previousOverflow;
      window.document.removeEventListener("keydown", handleKeyDown);
    };
  }, [confirmOpen, deleteMutation.isPending]);

  return (
    <>
      <button
        ref={triggerRef}
        type="button"
        onClick={openConfirmation}
        aria-label={`Delete ${document.filename}`}
        title={`Delete ${document.filename}`}
        className="rounded-lg p-2 text-slate-400 transition-colors hover:bg-red-50 hover:text-red-600"
      >
        <Trash2 className="h-4 w-4" />
      </button>

      {confirmOpen &&
        createPortal(
          <div
            className="fixed inset-0 z-[100] flex items-center justify-center bg-slate-950/50 p-4"
            onMouseDown={(event) => {
              if (event.target === event.currentTarget) {
                closeConfirmation();
              }
            }}
          >
            <div
              ref={dialogRef}
              role="dialog"
              aria-modal="true"
              aria-labelledby="delete-document-title"
              aria-describedby="delete-document-description"
              className="w-full max-w-md rounded-2xl bg-white p-6 text-left shadow-2xl sm:p-7"
            >
              {/* Delete icon */}
              <div className="flex h-12 w-12 items-center justify-center rounded-full bg-red-50">
                <Trash2 className="h-6 w-6 text-red-600" />
              </div>

              {/* Title */}
              <h2
                id="delete-document-title"
                className="mt-5 text-left text-xl font-semibold text-slate-900"
              >
                Delete document?
              </h2>

              {/* Description */}
              <p
                id="delete-document-description"
                className="mt-3 text-left text-sm leading-6 text-slate-600"
              >
                Are you sure you want to delete{" "}
                <span className="break-all font-semibold text-slate-900">
                  {document.filename}
                </span>
                ?
              </p>

              <p className="mt-2 text-left text-sm leading-6 text-slate-500">
                This will permanently remove the document and its indexed
                chunks from the knowledge base. This action cannot be undone.
              </p>

              {/* Backend error */}
              {deleteMutation.isError && (
                <div
                  role="alert"
                  className="mt-5 flex items-start gap-2 rounded-lg bg-red-50 p-3 text-sm text-red-700"
                >
                  <AlertCircle className="mt-0.5 h-4 w-4 shrink-0" />
                  <span>
                    {getDeleteErrorMessage(deleteMutation.error)}
                  </span>
                </div>
              )}

              {/* Actions */}
              <div className="mt-7 flex flex-wrap justify-end gap-3">
                <button
                  ref={cancelRef}
                  type="button"
                  onClick={closeConfirmation}
                  disabled={deleteMutation.isPending}
                  className="rounded-lg border border-slate-200 bg-white px-4 py-2.5 text-sm font-medium text-slate-700 hover:bg-slate-50 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-600 disabled:opacity-50"
                >
                  Cancel
                </button>

                <button
                  type="button"
                  onClick={() => deleteMutation.mutate()}
                  disabled={deleteMutation.isPending}
                  className="inline-flex items-center justify-center gap-2 rounded-lg bg-red-600 px-4 py-2.5 text-sm font-medium text-white hover:bg-red-700 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-red-600 disabled:cursor-not-allowed disabled:opacity-50"
                >
                  {deleteMutation.isPending ? (
                    <LoaderCircle className="h-4 w-4 animate-spin" />
                  ) : (
                    <Trash2 className="h-4 w-4" />
                  )}

                  {deleteMutation.isPending
                    ? "Deleting..."
                    : "Delete document"}
                </button>
              </div>
            </div>
          </div>,
          window.document.body
        )}
    </>
  );
}

function documentBodyOverflow(): string {
  return window.document.body.style.overflow;
}
